WITH total_sales_each_product_category AS 
(
SELECT SUM(od.unit_price * od.quantity * (1 - od.discount)) AS total_sales, c.category_id, c.category_name
FROM order_details AS od
JOIN products AS p
ON p.product_id = od.product_id
JOIN categories AS c
ON c.category_id = p.category_id
    
GROUP BY c.category_id, c.category_name
    )


SELECT category_id, category_name, total_sales,
total_sales/ SUM(total_sales) OVER() * 100 AS total_sales_percentage

FROM total_sales_each_product_category
