WITH cte AS ( SELECT SUM(od.unit_price * od.quantity * (1 - od.discount)) AS total_sales, DATE_TRUNC('month', o.order_date) AS month                                                                   
FROM orders as o
JOIN order_details as od
ON o.order_id = od.order_id
GROUP BY DATE_TRUNC('month', o.order_date) ),

lag_cte AS (
SELECT LAG(total_sales) OVER(ORDER BY month) AS total_sales_prev_month, cte.total_sales, cte.month
FROM cte

)


SELECT ((cte.total_sales::numeric - total_sales_prev_month ) * 100 / cte.total_sales) AS month_sales_growth_rate
FROM cte
JOIN lag_cte
ON cte.month = lag_cte.month
